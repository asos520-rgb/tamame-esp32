cmake_minimum_required(VERSION 3.16)
project(tamame_font_native C CXX)
if(NOT LVGL_SOURCE_DIR)
    message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the pinned SDK's LVGL source")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
set(COMP "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware-overlay/esp32/components/muse")
add_executable(tamame_font_test main.c "${COMP}/muse_i18n.c" "${COMP}/muse_ui_fonts.c" "${COMP}/fonts/muse_font_noto_tc_16.c")
target_compile_definitions(tamame_font_test PRIVATE LV_CONF_PATH="${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" LV_LVGL_H_INCLUDE_SIMPLE)
target_include_directories(tamame_font_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${COMP}" "${SDK_MUSE_INCLUDE_DIR}")
target_link_libraries(tamame_font_test PRIVATE lvgl m)

add_executable(tamame_wardrobe_test wardrobe.c "${COMP}/muse_i18n.c" "${COMP}/muse_ui_fonts.c" "${COMP}/fonts/muse_font_noto_tc_16.c")
target_compile_definitions(tamame_wardrobe_test PRIVATE CONFIG_MUSE_BOARD_WAVESHARE_S3_175C=1 LV_CONF_PATH="${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" LV_LVGL_H_INCLUDE_SIMPLE)
target_include_directories(tamame_wardrobe_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${COMP}" "${SDK_MUSE_INCLUDE_DIR}")
target_link_libraries(tamame_wardrobe_test PRIVATE lvgl m)
